BODY:
function(x, argument) {
  if (!is.numeric(x) || length(x) != 1L || is.na(x) ||
      !is.finite(x) || x < 0 || x > 1) {
    stop(
      sprintf("`%s` must be a single numeric value in [0, 1].", argument),
      call. = FALSE
    )
  }
  invisible(TRUE)
}
